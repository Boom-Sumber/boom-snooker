# 03 · 数据模型与接口

## 1. 表结构

### 1.1 用户资料

```sql
create table profiles (
  id          uuid primary key references auth.users(id) on delete cascade,
  nickname    text not null,
  avatar      text,
  role        text not null default 'viewer' check (role in ('admin','viewer')),
  created_at  timestamptz not null default now()
);

-- 注册后自动建档
create or replace function handle_new_user() returns trigger
language plpgsql security definer set search_path = public as $$
begin
  insert into profiles (id, nickname)
  values (new.id, coalesce(new.raw_user_meta_data->>'nickname', split_part(new.email, '@', 1)))
  on conflict (id) do nothing;
  return new;
end $$;

create trigger on_auth_user_created after insert on auth.users
for each row execute function handle_new_user();
```

### 1.2 彩球打法（参考数据）

```sql
create table balls (
  id           uuid primary key default gen_random_uuid(),
  color_code   text not null,          -- yellow/green/brown/blue/pink/black
  pocket_code  text not null,          -- top_left / bottom_left / top_middle / ...
  method       text not null,          -- 进袋方式（自由文本：直线、半台一库翻袋…）
  strength_code text not null,         -- full / half / ...
  sight        text not null,          -- 瞄准方法
  low_comp     text not null,          -- 低杆走位补偿
  high_comp    text not null,          -- 高杆走位补偿
  remark       text,                   -- 补充说明
  impact_image text not null,          -- Storage object path
  blue_image   text not null,
  route_image  text not null,
  state        int  not null default 1,
  updated_at   timestamptz not null default now(),
  updated_by   text                    -- 昵称，冗余存储避免 join
);

create index idx_balls_query on balls (color_code, pocket_code, method) where state = 1;
```

### 1.5 数据集版本

```sql
create table dataset_meta (
  id         int primary key default 1 check (id = 1),
  version    int not null default 1,
  updated_at timestamptz not null default now()
);
insert into dataset_meta (id, version) values (1, 1) on conflict do nothing;

-- 只有参考数据变更才 bump
create or replace function bump_dataset_version() returns trigger
language plpgsql as $$
begin
  update dataset_meta set version = version + 1, updated_at = now() where id = 1;
  return null;
end $$;

create trigger trg_balls_changed    after insert or update on balls        for each row execute function bump_dataset_version();
create trigger trg_advanced_changed after insert or update on advanced     for each row execute function bump_dataset_version();
create trigger trg_dict_changed     after insert or update on display_dict for each row execute function bump_dataset_version();
-- ⚠️ comments 表绝对不要加这个 trigger
```

### 1.6 评论

```sql
create table comments (
  id          uuid primary key default gen_random_uuid(),
  target_type text not null check (target_type in ('ball','advanced')),
  target_id   uuid not null,
  user_id     uuid not null references auth.users(id),
  content     text not null check (char_length(content) between 1 and 300),
  status      text not null default 'pending'
              check (status in ('pending','visible','rejected')),
  reviewed_by uuid references auth.users(id),
  reviewed_at timestamptz,
  review_note text,
  created_at  timestamptz not null default now(),
  deleted_at  timestamptz
);

create index idx_comments_public on comments (target_type, target_id, created_at desc)
  where deleted_at is null and status = 'visible';
create index idx_comments_pending on comments (created_at desc) where status = 'pending';
create index idx_comments_mine on comments (user_id, created_at desc);
```

## 2. 触发器（客户端无法绕过）

```sql
-- 强制 pending，忽略客户端传值
create or replace function force_comment_pending() returns trigger
language plpgsql as $$
begin
  new.status      := 'pending';
  new.user_id     := auth.uid();
  new.created_at  := now();
  new.reviewed_by := null;
  new.reviewed_at := null;
  new.review_note := null;
  new.deleted_at  := null;
  return new;
end $$;

create trigger trg_comment_pending before insert on comments
for each row execute function force_comment_pending();

-- 频率限制（pending 也计入，防止刷队列）
create or replace function check_comment_rate() returns trigger
language plpgsql as $$
begin
  if (select count(*) from comments
      where user_id = new.user_id and created_at > now() - interval '1 minute') >= 1 then
    raise exception 'RATE_LIMIT_MINUTE';
  end if;
  if (select count(*) from comments
      where user_id = new.user_id and created_at > now() - interval '1 day') >= 20 then
    raise exception 'RATE_LIMIT_DAY';
  end if;
  return new;
end $$;

create trigger trg_comment_rate before insert on comments
for each row execute function check_comment_rate();
```

## 3. RLS 策略

```sql
create or replace function is_admin() returns boolean
language sql security definer stable set search_path = public as $$
  select exists (select 1 from profiles where id = auth.uid() and role = 'admin');
$$;

-- ── 参考数据：登录用户只读，管理员可写 ──
alter table balls        enable row level security;
alter table advanced     enable row level security;
alter table display_dict enable row level security;
alter table dataset_meta enable row level security;
alter table profiles     enable row level security;
alter table comments     enable row level security;

create policy balls_read    on balls        for select using (auth.role() = 'authenticated');
create policy balls_write   on balls        for all    using (is_admin()) with check (is_admin());
create policy adv_read      on advanced     for select using (auth.role() = 'authenticated');
create policy adv_write     on advanced     for all    using (is_admin()) with check (is_admin());
create policy dict_read     on display_dict for select using (auth.role() = 'authenticated');
create policy meta_read     on dataset_meta for select using (auth.role() = 'authenticated');

create policy p_read   on profiles for select using (auth.role() = 'authenticated');
create policy p_update on profiles for update using (id = auth.uid());

-- ── 评论：核心是这条读策略 ──
create policy c_read on comments for select using (
  deleted_at is null and (
    status = 'visible'          -- 公开区
    or user_id = auth.uid()     -- 自己的（含 pending / rejected）
    or is_admin()               -- 管理员看全部
  )
);
create policy c_insert on comments for insert with check (user_id = auth.uid());
-- 不给 update / delete 策略：审核走 RPC，删除由管理员在 Dashboard 操作
```

## 4. RPC 清单

| 函数 | 参数 | 返回 | 用途 |
|---|---|---|---|
| `get_dataset_version()` | — | `int` | 启动时比对 |
| `get_dataset_snapshot()` | — | `jsonb` | 全量快照 |
| `upsert_ball(p jsonb, expected_version int)` | 字段 + 版本 | `{id, version}` | 新增/编辑 |
| `delete_ball(p_id uuid, expected_version int)` | id + 版本 | `void` | 逻辑删除 |
| `upsert_advanced(p jsonb, expected_version int)` | 同上 | `{id, version}` | 新增/编辑 |
| `delete_advanced(p_id uuid, expected_version int)` | 同上 | `void` | 逻辑删除 |
| `review_comment(p_id uuid, p_decision text, p_note text)` | — | `void` | 审核 |
| `get_comment_counts(p_type text, p_ids uuid[])` | — | `{target_id, cnt}[]` | 列表页评论数 |

### 4.1 快照

```sql
create or replace function get_dataset_version() returns int
language sql security definer stable as $$
  select version from dataset_meta where id = 1;
$$;

create or replace function get_dataset_snapshot() returns jsonb
language sql security definer stable as $$
  select jsonb_build_object(
    'version',  (select version from dataset_meta where id = 1),
    'balls',    coalesce((select jsonb_agg(to_jsonb(b) order by b.updated_at)
                            from balls b where b.state = 1), '[]'::jsonb),
    'advanced', coalesce((select jsonb_agg(to_jsonb(a) order by a.updated_at)
                            from advanced a where a.state = 1), '[]'::jsonb),
    'dict',     coalesce((select jsonb_agg(to_jsonb(d) order by d.kind, d.sort)
                            from display_dict d), '[]'::jsonb)
  );
$$;

grant execute on function get_dataset_version()  to authenticated;
grant execute on function get_dataset_snapshot() to authenticated;
```

### 4.2 写入（乐观并发）

```sql
create or replace function upsert_ball(p jsonb, expected_version int)
returns jsonb language plpgsql security definer as $$
declare cur int; new_id uuid;
begin
  if not is_admin() then raise exception 'FORBIDDEN'; end if;

  select version into cur from dataset_meta where id = 1;
  if expected_version is not null and cur <> expected_version then
    raise exception 'VERSION_CONFLICT';
  end if;

  new_id := coalesce(nullif(p->>'id','')::uuid, gen_random_uuid());

  insert into balls (id, color_code, pocket_code, method, strength_code, sight,
                     low_comp, high_comp, remark, impact_image, blue_image, route_image,
                     state, updated_at, updated_by)
  values (new_id, p->>'color_code', p->>'pocket_code', p->>'method', p->>'strength_code',
          p->>'sight', p->>'low_comp', p->>'high_comp', p->>'remark',
          p->>'impact_image', p->>'blue_image', p->>'route_image',
          1, now(), (select nickname from profiles where id = auth.uid()))
  on conflict (id) do update set
    color_code    = excluded.color_code,
    pocket_code   = excluded.pocket_code,
    method        = excluded.method,
    strength_code = excluded.strength_code,
    sight         = excluded.sight,
    low_comp      = excluded.low_comp,
    high_comp     = excluded.high_comp,
    remark        = excluded.remark,
    impact_image  = excluded.impact_image,
    blue_image    = excluded.blue_image,
    route_image   = excluded.route_image,
    state         = 1,
    updated_at    = now(),
    updated_by    = excluded.updated_by;

  return jsonb_build_object('id', new_id,
                            'version', (select version from dataset_meta where id = 1));
end $$;
```

> `delete_ball` / `upsert_advanced` / `delete_advanced` 结构相同：先 `is_admin()`，再校验 version，然后 `update ... set state = 0`。

### 4.3 审核

```sql
create or replace function review_comment(p_id uuid, p_decision text, p_note text default null)
returns void language plpgsql security definer as $$
begin
  if not is_admin() then raise exception 'FORBIDDEN'; end if;
  if p_decision not in ('visible','rejected') then raise exception 'BAD_DECISION'; end if;

  update comments
     set status = p_decision, reviewed_by = auth.uid(), reviewed_at = now(), review_note = p_note
   where id = p_id and status = 'pending';
end $$;

grant execute on function review_comment(uuid, text, text) to authenticated;
```

### 4.4 评论数

```sql
create or replace function get_comment_counts(p_type text, p_ids uuid[])
returns table (target_id uuid, cnt bigint)
language sql security definer stable as $$
  select c.target_id, count(*)
    from comments c
   where c.status = 'visible' and c.deleted_at is null
     and c.target_type = p_type and c.target_id = any(p_ids)
   group by c.target_id;
$$;

grant execute on function get_comment_counts(text, uuid[]) to authenticated;
```

## 5. 客户端调用约定

### 5.1 快照

```
GET  /rest/v1/rpc/get_dataset_version
GET  /rest/v1/rpc/get_dataset_snapshot     → { version, balls[], advanced[], dict[] }
```

本地落库流程：
```
下载 json → 写入 %LOCALAPPDATA%\Snooker\temp.db → 关闭连接 → File.Move(temp.db, cache.db, overwrite)
```
**必须用原子 rename**，不能直接写 cache.db——中途断网会损坏数据库。

### 5.2 评论查询

```
GET /rest/v1/comments
    ?target_type=eq.ball
    &target_id=eq.<uuid>
    &order=created_at.desc
    &limit=20
    &offset=0
    &select=id,user_id,content,status,created_at,review_note
```
返回结果里 `status` 已经由 RLS 过滤过，客户端只需按 status 渲染标签：
- `pending` → 橙色「审核中」
- `rejected` → 灰色「未通过」
- `visible` → 正常

### 5.3 发评论

```
POST /rest/v1/comments
body: { "target_type": "ball", "target_id": "<uuid>", "content": "..." }
```
**不要传** `user_id` / `status`——trigger 会强制覆盖。

### 5.4 审核

```
POST /rest/v1/rpc/review_comment
body: { "p_id": "<uuid>", "p_decision": "visible", "p_note": null }
```

### 5.5 错误码映射

PostgREST 会把 `raise exception 'XXX'` 的 message 原样返回，客户端按此映射：

| 服务端 message | 客户端提示 |
|---|---|
| `FORBIDDEN` | 无权限执行此操作 |
| `VERSION_CONFLICT` | 数据已更新，请刷新后重试 |
| `RATE_LIMIT_MINUTE` | 发得太快了，请稍后再试 |
| `RATE_LIMIT_DAY` | 今天评论太多了，明天再来 |
| `BAD_DECISION` | 非法操作 |

## 7. 注意

- `dataset_meta.version` 只在参考数据变更时 +1，**评论不影响**
- 所有时间字段用 `timestamptz`，客户端按本地时区显示
- 图片字段存的是 Storage object path（如 `balls/00b2e6b9-....webp`），客户端拼 URL：
  - 缩略图：`{SUPABASE_URL}/storage/v1/object/public/images/{path}?width=200&quality=70`
  - 原图：`{SUPABASE_URL}/storage/v1/object/public/images/{path}`
